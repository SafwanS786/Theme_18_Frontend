2️⃣ Add Animation to About_Main.jsx

We’ll animate the title and breadcrumb with a fade + slide.

import React from "react";
import { motion } from "framer-motion";
import "../../Styles/About_Style/About_first_part.css";
import About_Img from "../About_component/img/About_Img.png";

export default function About_Main() {
  return (
    <div>
      <div
        className="Main_About_us"
        style={{ backgroundImage: `url(${About_Img})` }}
      >
        <div className="ovr">
          <motion.h1
            className="title main_content"
            initial={{ opacity: 0, y: -50 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 1, ease: "easeOut" }}
          >
            About Us
          </motion.h1>

          <motion.p
            className="breadcrumb"
            initial={{ opacity: 0, y: 30 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 1, delay: 0.5, ease: "easeOut" }}
          >
            Home / About us
          </motion.p>
        </div>
      </div>
    </div>
  );
}

3️⃣ Add Animation to About_Future.jsx

We’ll animate the image and text so they slide in from opposite sides.

import React from "react";
import { motion } from "framer-motion";
import imgs_abt from "../About_component/img/abt.png";
import "../../Styles/About_Style/Future.css";

export default function About_Future() {
  return (
    <div className="container Abt_Future">
      {/* Image slides from left */}
      <motion.div
        className="abt_img_class"
        initial={{ opacity: 0, x: -100 }}
        whileInView={{ opacity: 1, x: 0 }}
        transition={{ duration: 1, ease: "easeOut" }}
        viewport={{ once: true }}
      >
        <img src={imgs_abt} alt="Future Plans" />
      </motion.div>

      {/* Text slides from right */}
      <motion.div
        className="text_content"
        initial={{ opacity: 0, x: 100 }}
        whileInView={{ opacity: 1, x: 0 }}
        transition={{ duration: 1, delay: 0.3, ease: "easeOut" }}
        viewport={{ once: true }}
      >
        <p className="abt_heading main_content">ABOUT US</p>
        <h1 className="bld">Building the Future With Precision and Purpose</h1>
        <div className="para_ftr">
          <p>
            At BuildGo, we are more than just contractors — we are partners in
            building your vision...
          </p>
          <p>
            We are a dedicated team of architects, engineers, and construction
            professionals committed...
          </p>
        </div>
      </motion.div>
    </div>
  );
}

🔟 Staggered Children Animation

If .ovr has multiple children, animate them one by one:

<motion.div
  className="ovr"
  initial="hidden"
  whileInView="visible"
  transition={{ staggerChildren: 0.2 }}
  viewport={{ once: true }}
>
  <motion.h1 variants={{
    hidden: { opacity: 0, y: -30 },
    visible: { opacity: 1, y: 0 }
  }} transition={{ duration: 0.6 }} />

  <motion.p variants={{
    hidden: { opacity: 0, y: 30 },
    visible: { opacity: 1, y: 0 }
  }} transition={{ duration: 0.6 }} />
</motion.div>
